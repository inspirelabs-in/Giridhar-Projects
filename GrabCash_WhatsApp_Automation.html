  const token = req.query['hub.verify_token'];
  const challenge = req.query['hub.challenge'];

  if (mode === 'subscribe' && token === VERIFY_TOKEN) {
    console.log('Webhook verified');
    return res.status(200).send(challenge);

// Incoming messages & status updates (POST)
app.post('/webhook', async (req, res) => {
  const body = req.body;

  if (body.object !== 'whatsapp_business_account') {
    return res.sendStatus(404);
  }

  for (const entry of body.entry) {
    for (const change of entry.changes) {
      const value = change.value;

      // Handle incoming messages
      if (value.messages) {
        for (const message of value.messages) {
          await handleIncomingMessage(message, value.metadata);
        }
      }

      // Handle message status updates
      if (value.statuses) {
        for (const status of value.statuses) {
          await handleStatusUpdate(status);
        }
      }
    }
  }

  res.sendStatus(200);
});

async function handleIncomingMessage(message, metadata) {
  const from = message.from; // sender phone number
  const msgType = message.type; // text, image, interactive, etc.

  switch (msgType) {
    case 'text':
      await processTextCommand(from, message.text.body);
      break;
    case 'interactive':
      await processButtonClick(from, message.interactive);
      break;
  }
}

  // status.status: 'sent', 'delivered', 'read', 'failed'
  const messageId = status.id;
  const recipientId = status.recipient_id;
  const timestamp = status.timestamp;
  const statusType = status.status;

import re
from dataclasses import dataclass
from typing import Optional, List

@dataclass
class DealSearchQuery:
    brand: Optional[str] = None
    category: Optional[str] = None
    max_price: Optional[int] = None
    min_discount: Optional[int] = None

def parse_deal_query(text: str) -> DealSearchQuery:
    """Parse natural language deal search queries.

    Examples:
        'Nike shoes under 2000' -> DealSearchQuery(brand='Nike', category='shoes', max_price=2000)
        'electronics above 50% off' -> DealSearchQuery(category='electronics', min_discount=50)
    """
    text_lower = text.lower()
    query = DealSearchQuery()

    # Extract price constraint
    price_match = re.search(r'under|below|less than|within)\s*(?:rs\.?|inr)?\s*(\d+)', text_lower)
    if price_match:
        query.max_price = int(price_match.group(1))

    # Extract discount constraint
    disc_match = re.search(r'(\d+)\s*%\s*(?:off|discount)', text_lower)
    if disc_match:
        query.min_discount = int(disc_match.group(1))

    # Extract brand (match against known brand list)
    known_brands = ['nike', 'adidas', 'samsung', 'apple', 'boat', 'oneplus',
                    'realme', 'prestige', 'bajaj', 'philips', 'himalaya']
    for brand in known_brands:
        if brand in text_lower:
            query.brand = brand.title()
            break

    # Extract category
    categories = {
        'shoes': ['shoes', 'footwear', 'sneakers', 'sandals'],
        'electronics': ['electronics', 'gadgets', 'tech'],
        'fashion': ['clothes', 'fashion', 'dress', 'shirt', 'jeans'],
        'mobile': ['mobile', 'phone', 'smartphone'],
        'kitchen': ['kitchen', 'cookware', 'appliance'],
        'beauty': ['beauty', 'skincare', 'makeup', 'cosmetics'],
    }
    for cat, keywords in categories.items():
        if any(kw in text_lower for kw in keywords):
            query.category = cat
            break

    return query

Node.js Message Queue with Rate Limiting
const Queue = require('bull');
const Redis = require('ioredis');

const redis = new Redis(process.env.REDIS_URL);

// Create message queue with rate limiting
const messageQueue = new Queue('whatsapp-messages', process.env.REDIS_URL, {
  limiter: {
    max: 70,          // Max 70 messages per second (below 80 API limit)
    duration: 1000,   // Per 1 second
  },
  defaultJobOptions: {
    attempts: 3,
    backoff: {
      type: 'exponential',
      delay: 5000,    // 5s, then 10s, then 20s retry
    },
    removeOnComplete: 1000,  // Keep last 1000 completed jobs
    removeOnFail: 5000,      // Keep last 5000 failed jobs
  }
});

// Process messages
messageQueue.process(async (job) => {
  const { phone, templateName, parameters, dealId, campaignId } = job.data;

  try {
    // Check if subscriber is still opted in
    const subscriber = await db.waSubscribers.findByPhone(phone);
    if (!subscriber || !subscriber.opted_in) {
      return { status: 'skipped', reason: 'not_opted_in' };
    }

    // Check daily marketing message limit (2 per user per day)
    if (templateName.startsWith('marketing_')) {
      const todayCount = await getMarketingCountToday(phone);
      if (todayCount >= 1) {  // We limit to 1 to be safe (2 is total across all businesses)
        return { status: 'skipped', reason: 'daily_limit_reached' };
      }
    }

    // Send via WhatsApp Cloud API
    const result = await sendWhatsAppTemplate(phone, templateName, parameters);

    // Log the message
    await db.waMessages.insert({
      wa_message_id: result.messages[0].id,
      subscriber_id: subscriber.id,
      phone: phone,
      template_name: templateName,
      template_category: getTemplateCategory(templateName),
      deal_id: dealId,
      campaign_id: campaignId,
      status: 'sent',
      sent_at: new Date(),
      cost: getTemplateCost(templateName),
    });

    return { status: 'sent', messageId: result.messages[0].id };

  } catch (error) {
    // Handle specific WhatsApp API errors
    if (error.code === 131026) {
      // Message undeliverable - user may have blocked or number invalid
      await db.waSubscribers.updateOptOut(phone, 'undeliverable');
      throw new Error('UNDELIVERABLE');
    }
    if (error.code === 130429) {
      // Rate limit hit - retry with backoff
      throw new Error('RATE_LIMITED');
    }
    throw error;
  }
});

// Campaign sender - batches messages for a campaign
async function sendCampaign(campaignId) {
  const campaign = await db.waCampaigns.findById(campaignId);
