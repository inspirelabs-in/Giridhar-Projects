def scan_category_deals(category_id, min_discount=30):
    """Scan a category for deals above minimum discount threshold."""
    request = SearchItemsRequest(
        partner_tag=PARTNER_TAG,
        partner_type=PartnerType.ASSOCIATES,
        browse_node_id=category_id,
        item_count=10,
        resources=[
            "ItemInfo.Title",
            "Offers.Listings.Price",
            "Offers.Listings.SavingBasis",
            "Offers.Listings.MerchantInfo",
            "Images.Primary.Large",
            "BrowseNodeInfo.BrowseNodes"
        ],
        sort_by="Price:LowToHigh",
        min_saving_percent=min_discount,
        marketplace="www.amazon.in",
        currency_of_preference="INR"
    )
    
    try:
        response = api.search_items(request)
        deals = []
        if response.search_result:
            for item in response.search_result.items:
                price = item.offers.listings[0].price.amount
                saving = item.offers.listings[0].saving_basis.amount
                discount = round((1 - price/saving) * 100, 1)
                deals.append({
                    "asin": item.asin,
                    "title": item.item_info.title.display_value,
                    "price": price,
                    "mrp": saving,
                    "discount": discount,
                    "url": item.detail_page_url
                })
        return deals
    except ApiException as e:
        print(f"PAAPI Error: {e}")
        return []

def get_deals_of_day():
    """Fetch Flipkart Deal of the Day listings."""
    url = f"{BASE_URL}/feeds/dealOfTheDay.json"
    response = requests.get(url, headers=headers)
    if response.status_code == 200:
        data = response.json()
        deals = []
        for product in data.get("dotdList", []):
            deal = {
                "title": product.get("title"),
                "url": product.get("url"),
                "image": product.get("imageUrls", {}).get("400x400"),
                "original_price": product.get("maximumRetailPrice", {}).get("amount"),
                "deal_price": product.get("flipkartSpecialPrice", {}).get("amount"),
                "discount": product.get("discountPercentage"),
                "availability": product.get("inStock"),
                "category": product.get("categoryPath")
            }
            deals.append(deal)
        return deals
    return []

def get_category_products(category_url, max_pages=5):

import sqlite3
import hashlib
from datetime import datetime
import requests
from bs4 import BeautifulSoup

class PriceTracker:
    """Custom price tracking database for Indian e-commerce."""
    
    def __init__(self, db_path="prices.db"):
        self.conn = sqlite3.connect(db_path)
        self._create_tables()
    
    def _create_tables(self):
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS price_history (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                product_id TEXT NOT NULL,
                merchant TEXT NOT NULL,
                price REAL NOT NULL,
                mrp REAL,
                discount_pct REAL,
                in_stock INTEGER DEFAULT 1,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
                UNIQUE(product_id, merchant, timestamp)
            )
        """)
        self.conn.execute("""
            CREATE INDEX IF NOT EXISTS idx_product_merchant 
            ON price_history(product_id, merchant)
        """)
        self.conn.commit()
    
    def record_price(self, product_id, merchant, price, mrp=None):
        """Record a price observation."""
        discount = round((1 - price/mrp) * 100, 1) if mrp and mrp > 0 else 0
        self.conn.execute(
            "INSERT OR IGNORE INTO price_history "
            "(product_id, merchant, price, mrp, discount_pct) "
            "VALUES (?, ?, ?, ?, ?)",
            (product_id, merchant, price, mrp, discount)
        )
        self.conn.commit()
    
    def get_historical_low(self, product_id, merchant):
        """Get the all-time lowest price for a product."""
        cursor = self.conn.execute(
            "SELECT MIN(price), timestamp FROM price_history "
            "WHERE product_id = ? AND merchant = ?",
            (product_id, merchant)
        )
        row = cursor.fetchone()
        return {"lowest_price": row[0], "date": row[1]} if row[0] else None
    
    def is_new_low(self, product_id, merchant, current_price):
        """Check if current price is a new all-time low."""
        hist = self.get_historical_low(product_id, merchant)
        if hist is None:
            return True  # First observation
        return current_price < hist["lowest_price"]
    
    def get_price_trend(self, product_id, merchant, days=30):
        """Get price trend over last N days."""
        cursor = self.conn.execute(
            "SELECT price, timestamp FROM price_history "
            "WHERE product_id = ? AND merchant = ? "
            "AND timestamp >= datetime('now', ?) "
            "ORDER BY timestamp ASC",
            (product_id, merchant, f"-{days} days")
        )
        return [{"price": r[0], "date": r[1]} for r in cursor.fetchall()]

Python — Deal Scoring Engine 
class DealScorer:
    """GrabCash Deal Scoring Algorithm v1.0"""
    
    # Category-specific base conversion rates
    CATEGORY_CVR = {
        "electronics": 3.5, "mobiles": 4.0, "fashion": 4.5,
        "beauty": 3.8, "home": 2.5, "grocery": 5.0,
        "travel": 1.8, "books": 3.0, "baby": 3.2,
        "health": 2.8, "finance": 6.0, "default": 3.0
    }
    
    def score(self, deal):
        """Calculate deal score (1-100)."""
        s1 = self._discount_score(deal["discount_pct"])     # 0-30
        s2 = self._historical_score(deal)                    # 0-20
        s3 = self._commission_score(deal["commission_inr"])  # 0-20
        s4 = self._conversion_score(deal)                    # 0-15
        s5 = self._urgency_score(deal)                       # 0-15
        
        total = s1 + s2 + s3 + s4 + s5
        return min(100, max(1, round(total)))
    
    def _discount_score(self, discount_pct):
        """Score 0-30 based on discount depth."""
        if discount_pct >= 80: return 30
        elif discount_pct >= 60: return 23 + (discount_pct - 60) * 0.35
        elif discount_pct >= 40: return 16 + (discount_pct - 40) * 0.35
        elif discount_pct >= 20: return 5 + (discount_pct - 20) * 0.55
        else: return 0
    
    def _historical_score(self, deal):
        """Score 0-20 based on historical price comparison."""
        if deal.get("is_new_low"): return 20
        pct_above_low = deal.get("pct_above_historical_low", 100)
        if pct_above_low <= 10: return 12
        elif pct_above_low <= 20: return 6
        return 0
    
    def _commission_score(self, commission_inr):
        """Score 0-20 based on absolute commission value."""
        if commission_inr >= 100: return 20
        elif commission_inr >= 50: return 16
        elif commission_inr >= 20: return 12
        elif commission_inr >= 5: return 8
        return 2
    
    def _conversion_score(self, deal):
        """Score 0-15 based on conversion probability."""
        base_cvr = self.CATEGORY_CVR.get(
            deal.get("category", "default"), 3.0
        )
        rating_bonus = (deal.get("rating", 0) - 3.5) * 2 if deal.get("rating", 0) > 3.5 else 0
        review_bonus = min(3, deal.get("review_count", 0) / 1000)
        score = min(15, base_cvr + rating_bonus + review_bonus)
        return round(score, 1)
    
    def _urgency_score(self, deal):
        """Score 0-15 based on time sensitivity."""
        if deal.get("is_flash_deal"): return 15
        if deal.get("stock_count", 999) < 50: return 12
        if deal.get("coupon_expires_hours", 999) < 24: return 10
        if deal.get("bank_offer_ending"): return 8
        return 3
    
    def get_tier(self, score):
        """Return quality tier for a given score."""
        if score >= 90: return "S"
        elif score >= 70: return "A"
        elif score >= 50: return "B"
        return "C"
    
    def get_action(self, score):
        """Return posting action for a given score."""
        tier = self.get_tier(score)
        actions = {
            "S": {"action": "INSTANT_POST", "delay": 0, "channels": "ALL"},
            "A": {"action": "PRIORITY_POST", "delay": 300, "channels": "CATEGORY"},
            "B": {"action": "BATCH_POST", "delay": 7200, "channels": "GENERAL"},
            "C": {"action": "SKIP", "delay": None, "channels": None}
        }
        return actions[tier]

import requests
from bs4 import BeautifulSoup
from datetime import datetime, timedelta
import json

class BankOfferTracker:
    """Track bank offers across major Indian e-commerce platforms."""
    
    BANK_OFFER_PAGES = {
        "amazon": "https://www.amazon.in/b?node=28727856031",
        "flipkart": "https://www.flipkart.com/offers-store",
        "myntra": "https://www.myntra.com/myntra-bank-offers"
    }
    
    def __init__(self):
        self.offers = []
        self.session = requests.Session()
        self.session.headers.update({
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                          "AppleWebKit/537.36 Chrome/125.0.0.0 Safari/537.36"
        })
    
    def parse_amazon_bank_offers(self, html):
        """Extract bank offers from Amazon offers page."""
        soup = BeautifulSoup(html, "html.parser")
        offers = []
        for card in soup.select(".a-cardui"):
            title = card.select_one(".a-text-bold")
            details = card.select_one(".a-text-normal")
            if title and "bank" in title.text.lower():
                offer = {
                    "bank": self._extract_bank_name(title.text),
                    "merchant": "amazon",
                    "description": title.text.strip(),
                    "details": details.text.strip() if details else "",
                    "min_purchase": self._extract_min_purchase(
                        details.text if details else ""
                    ),
                    "max_discount": self._extract_max_discount(
                        details.text if details else ""
                    ),
                    "card_type": self._detect_card_type(title.text),
                    "scraped_at": datetime.now().isoformat()
                }
                offers.append(offer)
        return offers
    
    def _extract_bank_name(self, text):
        banks = ["HDFC", "ICICI", "SBI", "Axis", "Kotak",
                 "IDFC", "Amex", "RBL", "BOB", "IndusInd"]
        for bank in banks:
            if bank.lower() in text.lower():
                return bank
        return "Unknown"
    
    def _detect_card_type(self, text):
        text_lower = text.lower()
        if "credit" in text_lower: return "Credit Card"
        if "debit" in text_lower: return "Debit Card"
        if "emi" in text_lower: return "EMI"
        return "All Cards"
    
    def _extract_min_purchase(self, text):
        """Extract minimum purchase amount from offer text."""
        import re
        match = re.search(r'min\.?\s*(?:purchase|order|txn).*?(\d[\d,]+)', text, re.I)
        if match:
            return int(match.group(1).replace(",", ""))
        return 0
    
    def _extract_max_discount(self, text):
        """Extract maximum discount from offer text."""
        import re
        match = re.search(r'(?:upto?|max\.?|maximum)\s*(?:Rs\.?|INR|₹)?\s*(\d[\d,]+)', text, re.I)
        if match:
            return int(match.group(1).replace(",", ""))
        return 0
